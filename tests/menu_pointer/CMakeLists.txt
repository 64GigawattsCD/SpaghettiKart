cmake_minimum_required(VERSION 3.20)
project(menu_pointer_tests C)
enable_testing()
add_executable(menu_pointer_test menu_pointer_test.c ../../src/menu_pointer.c)
set_property(TARGET menu_pointer_test PROPERTY C_STANDARD 11)
add_test(NAME menu_pointer COMMAND menu_pointer_test)

# Compile the actual art helpers with recording primitives, without linking the game.
set(art_source "${CMAKE_CURRENT_SOURCE_DIR}/../../src/race_mods_menu.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${art_source}")
file(READ "${art_source}" art)
string(FIND "${art}" "void race_mods_draw_native_focus(" art_begin)
string(FIND "${art}" "// Six touch-sized rows" art_end)
if(art_begin LESS 0 OR art_end LESS art_begin)
    message(FATAL_ERROR "Native menu art helper section not found")
endif()
math(EXPR art_length "${art_end} - ${art_begin}")
string(SUBSTRING "${art}" ${art_begin} ${art_length} art)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/native_menu_art.inc" "${art}")
add_executable(native_menu_art_test native_menu_art_test.c)
target_include_directories(native_menu_art_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
set_property(TARGET native_menu_art_test PROPERTY C_STANDARD 11)
add_test(NAME native_menu_art COMMAND native_menu_art_test)
