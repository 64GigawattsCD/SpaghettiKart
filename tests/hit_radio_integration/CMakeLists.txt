cmake_minimum_required(VERSION 3.20)
project(hit_radio_integration C)
enable_testing()
set(source "${CMAKE_CURRENT_SOURCE_DIR}/../../src/audio/external.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
file(READ "${source}" audio)
string(FIND "${audio}" "void play_hit_radio_taunt(" begin)
string(FIND "${audio}" "void func_800C9250(" end)
if(begin LESS 0 OR end LESS begin)
    message(FATAL_ERROR "Radio audio section not found")
endif()
math(EXPR length "${end} - ${begin}")
string(SUBSTRING "${audio}" ${begin} ${length} radio)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/radio_audio.inc" "${radio}")
add_executable(hit_radio_audio_test hit_radio_audio_test.c)
target_include_directories(hit_radio_audio_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
set_property(TARGET hit_radio_audio_test PROPERTY C_STANDARD 11)
add_test(NAME hit_radio_audio_routing COMMAND hit_radio_audio_test)
